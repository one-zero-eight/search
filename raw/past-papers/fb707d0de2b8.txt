Database Schema A
CREATE TABLE Department (
departmentName VARCHAR(50),
budget INTEGER,
PRIMARY KEY (departmentName)
);
CREATE TABLE Location (
locationId VARCHAR(3),
city VARCHAR(10),
country VARCHAR (10),
PRIMARY KEY (locationId)
);
CREATE TABLE Employee (
employeeId INTEGER,
name VARCHAR(30),
jobTitle VARCHAR(15),
departmentName VARCHAR (50)
NOT NULL,
PRIMARY KEY (employeeId),
FOREIGN KEY (departmentName)
REFERENCES Department
);
CREATE TABLE Project (
projectCode VARCHAR(5),
projectType VARCHAR(10),
teamSize INTEGER,
PRIMARY KEY (projectCode)
);
CREATE TABLE Resource (
resourceId INTEGER,
resourceName VARCHAR(50),
locationId VARCHAR(3),
PRIMARY KEY(resourceId),
FOREIGN KEY (locationId)
REFERENCES Location
);
CREATE TABLE Assignment (
assignmentId INTEGER,
startDate TIMESTAMP,
endDate TIMESTAMP,
PRIMARY KEY (assignmentId)
);
CREATE TABLE Skill (
skillId INTEGER,
skillName VARCHAR(20),
PRIMARY KEY (skillId)
);

CREATE TABLE​
ResourceAllocation (
resourceId INTEGER,
projectCode VARCHAR(5),
assignmentId INTEGER,
allocationPercentage INTEGER,
PRIMARY KEY (resourceId,
projectCode, assignmentId),
FOREIGN KEY (resourceId)
REFERENCES Resource,
FOREIGN KEY (projectCode)
REFERENCES Project,
FOREIGN KEY (assignmentId)
REFERENCES Assignment
);
CREATE TABLE ProjectLocation (
projectCode VARCHAR(5),
locationId VARCHAR(3),
PRIMARY KEY (projectCode,
locationId),
FOREIGN KEY (projectCode)
REFERENCES Project,
FOREIGN KEY (locationId)
REFERENCES Location
);
CREATE TABLE WorksOn (
employeeId INTEGER,
projectCode VARCHAR(5),
assignmentId INTEGER,
hoursWorked INTEGER,
PRIMARY KEY (employeeId,
projectCode, assignmentId),
FOREIGN KEY (employeeId)
REFERENCES Employee,
FOREIGN KEY (projectCode)
REFERENCES Project,
FOREIGN KEY (assignmentId)
REFERENCES Assignment
);
CREATE TABLE TaskAssignment (
employeeId INTEGER,
taskId INTEGER,
assignmentId INTEGER,
role VARCHAR(20),
PRIMARY KEY (employeeId,
taskId, assignmentId),
FOREIGN KEY (employeeId)
REFERENCES Employee,
FOREIGN KEY (taskId)
REFERENCES Task,
FOREIGN KEY (assignmentId)
REFERENCES Assignment
);

CREATE TABLE EmployeeSkill (
employeeId INTEGER,
skillId INTEGER,
proficiencyLevel VARCHAR(10),
PRIMARY KEY (employeeId,
skillId),
FOREIGN KEY (employeeId)
REFERENCES Employee,
FOREIGN KEY (skillId)
REFERENCES Skill
);
CREATE TABLE Task (
taskId INTEGER,
taskDescription VARCHAR(100),
projectCode VARCHAR(5),
PRIMARY KEY (taskId),
FOREIGN KEY (projectCode)
REFERENCES Project
);
CREATE TABLE Manager (
employeeId INTEGER,
bonus DECIMAL (10, 2), -Additional attribute specific to
Managers
PRIMARY KEY (employeeId),
FOREIGN KEY (employeeId)
REFERENCES Employee (employeeId)
);
CREATE TABLE Engineer (
employeeId INTEGER,
specialization VARCHAR(50),
-- Additional attribute specific
to Engineers
PRIMARY KEY (employeeId),
FOREIGN KEY (employeeId)

CREATE TABLE SoftwareEngineer (
employeeId INTEGER,
programmingLanguage

SELECT 1

WHERE w.employeeId = e.employeeId​

SELECT 1
